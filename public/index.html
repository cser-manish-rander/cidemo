// ❌ VULNERABLE APPROACH (Allows XSS)
app.get('/reflect', (req, res) => {
    res.send(`<div>${req.query.val}</div>`); 
});

//  SECURE APPROACH (Escapes special characters)
const escapeHtml = require('escape-html');
app.get('/reflect', (req, res) => {
    res.send(`<div>${escapeHtml(req.query.val)}</div>`);
});

// ❌ VULNERABLE APPROACH (Concatenated Execution)
exec(`ping -c 3 ${req.query.ip}`);

//  SECURE APPROACH (Strict RegEx + Process Spawn)
const ipv4Regex = /^(?:[0-9]{1,3}\.){3}[0-9]{1,3}$/;
if (!ipv4Regex.test(req.query.ip)) {
    return res.status(400).send('Invalid Format');
